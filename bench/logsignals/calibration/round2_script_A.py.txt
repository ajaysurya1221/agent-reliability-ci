def norm_pair(rnd, rho):
    z1 = rnd.gauss(0.0, 1.0)
    z2 = rnd.gauss(0.0, 1.0)
    return z1, rho * z1 + math.sqrt(1.0 - rho * rho) * z2


def series_indep(rnd, n, cc):
    e = []
    f = []
    for _ in range(n):
        a, b = norm_pair(rnd, cc)
        e.append(a)
        f.append(b)
    return e, f


def series_ar(rnd, n, pu, pv, cc):
    su = math.sqrt(1.0 - pu * pu)
    sv = math.sqrt(1.0 - pv * pv)
    a, b = norm_pair(rnd, cc)
    cu, cv = a, b                      # stationary start, unit variance
    e = [cu]
    f = [cv]
    for _ in range(n - 1):
        a, b = norm_pair(rnd, cc)
        cu = pu * cu + su * a
        cv = pv * cv + sv * b
        e.append(cu)
        f.append(cv)
    return e, f


def series_regime(rnd, n):
    p = 1.0 / REGIME_MEAN_LEN
    k1 = math.sqrt(1.0 - REGIME_NOISE)
    k2 = math.sqrt(REGIME_NOISE)

    def one():
        out = []
        lvl = rnd.gauss(0.0, 1.0)
        for _ in range(n):
            out.append(k1 * lvl + k2 * rnd.gauss(0.0, 1.0))
            if rnd.random() < p:
                lvl = rnd.gauss(0.0, 1.0)
        return out

    return one(), one()


def series_rw(rnd, n):
    sc = 1.0 / math.sqrt(n)            # keeps the pooled scale comparable

    def one():
        out = []
        cur = 0.0
        for _ in range(n):
            cur += rnd.gauss(0.0, 1.0)
            out.append(cur * sc)
        return out

    return one(), one()


def within_series(rnd, n, cfg):
    kind = cfg["kind"]
    if kind == "indep":
        return series_indep(rnd, n, cfg.get("cc", 0.0))
    if kind == "ar":
        return series_ar(rnd, n, cfg["pu"], cfg["pv"], cfg.get("cc", 0.0))
    if kind == "regime":
        return series_regime(rnd, n)
    if kind == "rw":
        return series_rw(rnd, n)
    raise ValueError(kind)

def gen_lengths(rnd):
    out = []
    for n in SIZES:
        row = []
        for _ in range(n):
            d = int(round(LEN_MEDIAN * math.exp(LEN_SIGMA * rnd.gauss(0.0, 1.0))))
            if d < LEN_MIN:
                d = LEN_MIN
            row.append(d)
        out.append(row)
    return out


def gen_models(rnd):
    """1 to 3 names per repository, in contiguous runs over the index order.

    About half of the larger repositories (n >= 11) carry one name only.
    """
    out = []
    for n in SIZES:
        if n == 1:
            out.append([rnd.choice(MODEL_POOL)])
            continue
        if n >= 11:
            k = 1 if rnd.random() < 0.5 else rnd.choice((2, 3))
        else:
            k = 1 if rnd.random() < 0.5 else 2
        k = min(k, n)
        names = rnd.sample(MODEL_POOL, k)
        if k == 1:
            out.append([names[0]] * n)
            continue
        cuts = sorted(rnd.sample(range(1, n), k - 1))
        row = []
        bounds = [0] + cuts + [n]
        for j in range(k):
            row.extend([names[j]] * (bounds[j + 1] - bounds[j]))
        out.append(row)
    return out


def contiguous_runs(rnd, n, k):
    """Split 0..n-1 into k contiguous runs; returns a list of run labels."""
    k = min(k, n)
    if k == 1:
        return [0] * n
    cuts = sorted(rnd.sample(range(1, n), k - 1))
    bounds = [0] + cuts + [n]
    lab = []
    for j in range(k):
        lab.extend([j] * (bounds[j + 1] - bounds[j]))
    return lab

def share_of_p(p):
    ks = SHARE_KNOTS
    for i in range(1, len(ks)):
        if p <= ks[i][0]:
            p0, s0 = ks[i - 1]
            p1, s1 = ks[i]
            if p1 == p0:
                return s1
            return s0 + (s1 - s0) * (p - p0) / (p1 - p0)
    return ks[-1][1]


def map_outcome(latv):
    """Monotone map of the pooled outcome latent to the fixed outcome marginal."""
    order = sorted(range(len(latv)), key=lambda i: (latv[i], i))
    out = [0] * len(latv)
    for pos, i in enumerate(order):
        if pos < N_OUT_ZERO:
            out[i] = 0
        elif pos >= N_OUT_ZERO + N_OUT_MID:
            out[i] = 100
        else:
            j = pos - N_OUT_ZERO
            out[i] = 1 + int(round(98.0 * j / float(N_OUT_MID - 1)))
    return out


def map_signal(latu, dlist):
    """Monotone map of the pooled signal latent to a share k/d."""
    n = len(latu)
    order = sorted(range(n), key=lambda i: (latu[i], i))
    out = [0.0] * n
    for pos, i in enumerate(order):
        if pos < N_SIG_ZERO:
            out[i] = 0.0
            continue
        p = (pos + 0.5) / float(n)
        s = share_of_p(p)
        d = dlist[i]
        k = int(round(s * d))
        if k < 1:
            k = 1
        if k >= d:
            k = d - 1
        out[i] = k / float(d)
    return out

class Data(object):
    __slots__ = ("sig", "out", "dlen", "model", "rep", "nrep")


def gen_dataset(cfg, seed):
    rnd = random.Random(seed)
    nrep = len(SIZES)

    rea = []
    reb = []
    for _ in range(nrep):
        a, b = norm_pair(rnd, REP_CORR)
        rea.append(a)
        reb.append(b)

    lengths = gen_lengths(rnd)
    models = gen_models(rnd)

    all_ld = [math.log(d) for row in lengths for d in row]
    mld = sum(all_ld) / len(all_ld)
    vld = sum((x - mld) ** 2 for x in all_ld) / len(all_ld)
    sld = math.sqrt(vld) if vld > 0 else 1.0

    meff = {}
    if cfg.get("modeleff"):
        for name in MODEL_POOL:
            meff[name] = norm_pair(rnd, MODEL_CORR)

    sa = math.sqrt(REP_SHARE)
    sw = math.sqrt(1.0 - REP_SHARE)

    latu = []
    latv = []
    for i, n in enumerate(SIZES):
        e, f = within_series(rnd, n, cfg)

        if cfg.get("trend") and i < BIG_TREND:
            sd = math.sqrt((n * n - 1) / 12.0) if n > 1 else 1.0
            k1 = math.sqrt(1.0 - TREND_SHARE)
            k2 = math.sqrt(TREND_SHARE)
            ct = [((t - (n - 1) / 2.0) / sd) if sd > 0 else 0.0 for t in range(n)]
            e = [k1 * e[t] + k2 * ct[t] for t in range(n)]
            f = [k1 * f[t] + k2 * ct[t] for t in range(n)]

        if cfg.get("people") and i < BIG_PEOPLE:
            lab = contiguous_runs(rnd, n, 3)
            pe = [norm_pair(rnd, PEOPLE_CORR) for _ in range(3)]
            k1 = math.sqrt(1.0 - PEOPLE_SHARE)
            k2 = math.sqrt(PEOPLE_SHARE)
            e = [k1 * e[t] + k2 * pe[lab[t]][0] for t in range(n)]
            f = [k1 * f[t] + k2 * pe[lab[t]][1] for t in range(n)]

        if cfg.get("lengtheff"):
            lam = cfg["lengtheff"]
            k1 = math.sqrt(1.0 - lam)
            k2 = math.sqrt(lam)
            zd = [(math.log(lengths[i][t]) - mld) / sld for t in range(n)]
            e = [k1 * e[t] + k2 * zd[t] for t in range(n)]
            f = [k1 * f[t] + k2 * zd[t] for t in range(n)]

        if cfg.get("modeleff"):
            lam = cfg["modeleff"]
            k1 = math.sqrt(1.0 - lam)
            k2 = math.sqrt(lam)
            e = [k1 * e[t] + k2 * meff[models[i][t]][0] for t in range(n)]
            f = [k1 * f[t] + k2 * meff[models[i][t]][1] for t in range(n)]

        for t in range(n):
            latu.append(sa * rea[i] + sw * e[t])
            latv.append(sa * reb[i] + sw * f[t])

    dflat = [d for row in lengths for d in row]
    mflat = [m for row in models for m in row]
    rep = []
    for i, n in enumerate(SIZES):
        rep.extend([i] * n)

    dat = Data()
    dat.out = map_outcome(latv)
    dat.sig = map_signal(latu, dflat)
    dat.dlen = dflat
    dat.model = mflat
    dat.rep = rep
    dat.nrep = nrep
    return dat

def midranks2(vals):
    """Twice the midrank of each value, as integers."""
    n = len(vals)
    order = sorted(range(n), key=lambda i: vals[i])
    res = [0] * n
    i = 0
    while i < n:
        j = i
        vi = vals[order[i]]
        while j + 1 < n and vals[order[j + 1]] == vi:
            j += 1
        m2 = i + j + 2                 # 2 * ((i+1)+(j+1))/2
        for k in range(i, j + 1):
            res[order[k]] = m2
        i = j + 1
    return res


class Block(object):
    """One permutation block: its sessions in index order."""
    __slots__ = ("n", "x2", "y2", "c", "const", "obs", "w", "rho",
                 "shifts", "chunks1", "chunks2")


def make_block(sig, out, idxs):
    """Build a weight-carrying block, or return None."""
    n = len(idxs)
    if n < 2:
        return None
    xs = [sig[i] for i in idxs]
    ys = [out[i] for i in idxs]
    if xs.count(xs[0]) == n or ys.count(ys[0]) == n:
        return None
    x2 = midranks2(xs)
    y2 = midranks2(ys)
    sx = sum(x2)
    sy = sum(y2)
    xb = sx / float(n)
    yb = sy / float(n)
    vx = sum((a - xb) ** 2 for a in x2)
    vy = sum((a - yb) ** 2 for a in y2)
    if vx <= 0.0 or vy <= 0.0:
        return None
    b = Block()
    b.n = n
    b.x2 = x2
    b.y2 = y2
    b.w = n - 1
    b.c = (n - 1) / math.sqrt(vx * vy)
    b.const = yb * sx
    b.obs = b.c * (sum(map(mul, x2, y2)) - b.const)
    b.rho = b.obs / float(b.w)
    b.shifts = None
    b.chunks1 = None
    b.chunks2 = None
    return b


def blockset_by_key(sig, out, keys):
    """Group sessions by key, keeping each block's sessions in index order."""
    groups = {}
    order = []
    for i, k in enumerate(keys):
        if k not in groups:
            groups[k] = []
            order.append(k)
        groups[k].append(i)
    blocks = []
    for k in order:
        b = make_block(sig, out, groups[k])
        if b is not None:
            blocks.append(b)
    return blocks


def bs_stats(blocks):
    """Return (T, W, m, rho_bar)."""
    t = 0.0
    w = 0
    for b in blocks:
        t += b.obs
        w += b.w
    if w == 0:
        return 0.0, 0, 0, float("nan")
    return t, w, len(blocks), t / float(w)

def terciles_within(idxs, dlen):
    """Tercile label per session of one group: sort by length, ties by index."""
    n = len(idxs)
    srt = sorted(idxs, key=lambda i: (dlen[i], i))
    c1 = n // 3
    c2 = (2 * n) // 3
    lab = {}
    for pos, i in enumerate(srt):
        if pos < c1:
            lab[i] = 0
        elif pos < c2:
            lab[i] = 1
        else:
            lab[i] = 2
    return lab


def keys_repo_len(dat):
    """repository x length tercile, terciles cut inside the repository."""
    groups = {}
    for i, r in enumerate(dat.rep):
        groups.setdefault(r, []).append(i)
    keys = [None] * len(dat.rep)
    for r, idxs in groups.items():
        lab = terciles_within(idxs, dat.dlen)
        for i in idxs:
            keys[i] = (r, lab[i])
    return keys


def keys_repo_model(dat):
    return [(r, dat.model[i]) for i, r in enumerate(dat.rep)]


def keys_joint(dat):
    """repository x model name x length tercile, terciles cut inside repo x model."""
    groups = {}
    for i, r in enumerate(dat.rep):
        groups.setdefault((r, dat.model[i]), []).append(i)
    keys = [None] * len(dat.rep)
    for g, idxs in groups.items():
        lab = terciles_within(idxs, dat.dlen)
        for i in idxs:
            keys[i] = (g[0], g[1], lab[i])
    return keys

def prep_shifts(blocks):
    for b in blocks:
        if b.shifts is None:
            x2 = b.x2
            y2 = b.y2
            n = b.n
            c = b.c
            const = b.const
            vals = []
            for k in range(n):
                xs = x2[k:] + x2[:k]
                vals.append(c * (sum(map(mul, xs, y2)) - const))
            b.shifts = vals


def prep_chunks(blocks, which):
    for b in blocks:
        cur = b.chunks1 if which == 1 else b.chunks2
        if cur is not None:
            continue
        n = b.n
        if which == 1:
            L = int(math.ceil(math.sqrt(n)))
        else:
            L = int(math.ceil(n ** (2.0 / 3.0)))
        if L < 1:
            L = 1
        ch = [b.x2[i:i + L] for i in range(0, n, L)]
        if which == 1:
            b.chunks1 = ch
        else:
            b.chunks2 = ch


def pval_S(blocks, tobs, B, rnd):
    shuffle = rnd.shuffle
    lim = abs(tobs) - TOL
    hit = 0
    prep = [(list(b.x2), b.y2, b.c, b.const) for b in blocks]
    for _ in range(B):
        t = 0.0
        for x2, y2, c, const in prep:
            shuffle(x2)
            t += c * (sum(map(mul, x2, y2)) - const)
        if abs(t) >= lim:
            hit += 1
    return (1 + hit) / float(B + 1)


def pval_C(blocks, tobs, B, rnd):
    prep_shifts(blocks)
    lim = abs(tobs) - TOL
    hit = 0
    rr = rnd.randrange
    tab = [(b.shifts, b.n) for b in blocks]
    for _ in range(B):
        t = 0.0
        for vals, n in tab:
            t += vals[rr(n)]
        if abs(t) >= lim:
            hit += 1
    return (1 + hit) / float(B + 1)


def pval_F(blocks, tobs, B, rnd):
    lim = abs(tobs) - TOL
    hit = 0
    obs = [b.obs for b in blocks]
    rr = rnd.random
    for _ in range(B):
        t = 0.0
        for o in obs:
            t += o if rr() < 0.5 else -o
        if abs(t) >= lim:
            hit += 1
    return (1 + hit) / float(B + 1)


def _pval_block_perm(blocks, tobs, B, rnd, which):
    prep_chunks(blocks, which)
    shuffle = rnd.shuffle
    lim = abs(tobs) - TOL
    hit = 0
    prep = []
    for b in blocks:
        ch = b.chunks1 if which == 1 else b.chunks2
        prep.append((list(ch), b.y2, b.c, b.const))
    for _ in range(B):
        t = 0.0
        for ch, y2, c, const in prep:
            shuffle(ch)
            s = 0
            pos = 0
            for part in ch:
                lp = len(part)
                s += sum(map(mul, part, y2[pos:pos + lp]))
                pos += lp
            t += c * (s - const)
        if abs(t) >= lim:
            hit += 1
    return (1 + hit) / float(B + 1)


def pval_K(blocks, tobs, B, rnd):
    return _pval_block_perm(blocks, tobs, B, rnd, 1)


def pval_K2(blocks, tobs, B, rnd):
    return _pval_block_perm(blocks, tobs, B, rnd, 2)


PVAL = {"S": pval_S, "C": pval_C, "F": pval_F, "K": pval_K, "K2": pval_K2}

def jack_interval(blocks):
    """Cluster jackknife over contributing blocks, with the 1/sqrt(W) floor."""
    t, w, m, rb = bs_stats(blocks)
    if m < 2:
        return None
    loo = []
    for b in blocks:
        wd = w - b.w
        if wd <= 0:
            return None
        loo.append((t - b.obs) / float(wd))
    mean = sum(loo) / m
    ss = sum((x - mean) ** 2 for x in loo)
    se = math.sqrt((m - 1) / float(m) * ss)
    floor = 1.0 / math.sqrt(w)
    half = t975(m - 1) * max(se, floor)
    lo = rb - half
    hi = rb + half
    if lo < -1.0:
        lo = -1.0
    if hi > 1.0:
        hi = 1.0
    return rb, lo, hi, m, w, se, floor

def seed_for(tag, i):
    h = hashlib.sha256(("A|" + tag + "|" + str(i)).encode("utf-8")).digest()
    return int.from_bytes(h[:8], "big")


def run_primary_one(scen, i, B, tests):
    dat = gen_dataset(SCEN[scen], seed_for("prim|" + scen, i))
    blocks = blockset_by_key(dat.sig, dat.out, keys_repo(dat))
    t, w, m, rb = bs_stats(blocks)
    iv = jack_interval(blocks)
    res = {"rho_bar": rb, "W": w, "m": m}
    if iv is None:
        res["lo"] = res["hi"] = float("nan")
        res["floor_binds"] = 0
    else:
        res["lo"] = iv[1]
        res["hi"] = iv[2]
        res["floor_binds"] = 1 if iv[5] <= iv[6] else 0
    for name in tests:
        rnd = random.Random(seed_for("perm|" + scen + "|" + name, i))
        res[name] = PVAL[name](blocks, t, B, rnd)
    return res


def run_truth_one(scen, i):
    dat = gen_dataset(SCEN[scen], seed_for("truth|" + scen, i))
    blocks = blockset_by_key(dat.sig, dat.out, keys_repo(dat))
    _, w, _, rb = bs_stats(blocks)
    return rb


def run_gate_one(scen, i, B, tests):
    dat = gen_dataset(SCEN[scen], seed_for("gate|" + scen, i))
    sets = {
        "P": blockset_by_key(dat.sig, dat.out, keys_repo(dat)),
        "J": blockset_by_key(dat.sig, dat.out, keys_joint(dat)),
        "L": blockset_by_key(dat.sig, dat.out, keys_repo_len(dat)),
        "M": blockset_by_key(dat.sig, dat.out, keys_repo_model(dat)),
    }
    res = {}
    for key, blocks in sets.items():
        t, w, m, rb = bs_stats(blocks)
        res[key + "_W"] = w
        res[key + "_m"] = m
        res[key + "_rho"] = rb
        for name in tests:
            rnd = random.Random(seed_for("gperm|" + scen + "|" + key + "|" + name, i))
            res[key + "_" + name] = PVAL[name](blocks, t, B, rnd) if m > 0 else 1.0
    return res

def _w_primary(arg):
    scen, lo, hi, B, tests = arg
    return [run_primary_one(scen, i, B, tests) for i in range(lo, hi)]


def _w_truth(arg):
    scen, lo, hi = arg

    scen, lo, hi, B, tests = arg
    return [run_gate_one(scen, i, B, tests) for i in range(lo, hi)]


def chunks(scen, n, extra, size):
    out = []
    i = 0
    while i < n:
        j = min(n, i + size)
        out.append((scen, i, j) + tuple(extra))
        i = j
    return out


def run_pool(fn, tasks, nproc):
    if nproc <= 1:
        res = []
        for t in tasks:
            res.extend(fn(t))
        return res
    import multiprocessing as mp
    ctx = mp.get_context("fork")
    with ctx.Pool(nproc) as pool:
        res = []
        for part in pool.map(fn, tasks, chunksize=1):
            res.extend(part)
        return res

def enum_T_values(blocks):
    """All T values over the product of within-block permutations."""
    from itertools import permutations, product
    per = []
    for b in blocks:
        vals = []
        seen = []
        for p in permutations(range(b.n)):
            xs = [b.x2[k] for k in p]
            vals.append((p, b.c * (sum(map(mul, xs, b.y2)) - b.const)))
        per.append(vals)
    out = []
    for combo in product(*per):
        out.append((tuple(c[0] for c in combo), sum(c[1] for c in combo)))
    return out


def check_tiny(log):
    """S's Monte Carlo p-value against full enumeration on 3 repositories 4,3,2."""
    rnd = random.Random(seed_for("check|tiny", 0))
    # a small synthetic set built by hand; shares k/d and outcomes 0..100
    sig = [3 / 40.0, 9 / 40.0, 0.0, 17 / 40.0,
           5 / 30.0, 5 / 30.0, 1 / 30.0,
           7 / 25.0, 2 / 25.0]
    out = [0, 100, 44, 12,
           100, 0, 61,
           0, 100]
    keys = [0, 0, 0, 0, 1, 1, 1, 2, 2]
    blocks = blockset_by_key(sig, out, keys)
    t, w, m, rb = bs_stats(blocks)
    allv = enum_T_values(blocks)
    lim = abs(t) - TOL
    hit = sum(1 for _, v in allv if abs(v) >= lim)
    p_exact = hit / float(len(allv))
    B = 200000
    p_mc = pval_S(blocks, t, B, random.Random(seed_for("check|tinymc", 0)))
    mc_err = math.sqrt(max(p_exact * (1 - p_exact), 1e-12) / B)
    ok = abs(p_mc - p_exact) <= 4.0 * mc_err + 1.0 / (B + 1)
    log("CHECK 1  S against full enumeration, 3 blocks of sizes 4, 3, 2")
    log("  arrangements enumerated : %d" % len(allv))
    log("  T observed              : %.12f   rho_bar %.6f  W %d  m %d" % (t, rb, w, m))
    log("  p exact (enumeration)   : %.6f" % p_exact)
    log("  p Monte Carlo B=%d  : %.6f" % (B, p_mc))
    log("  Monte Carlo error       : %.6f" % mc_err)
    log("  agree within error      : %s" % ("YES" if ok else "NO"))
    return ok


def check_ties_equal(log):
    """Mathematically equal statistics from different arrangements count as equal."""
    # one block of 5 sessions; the signal has a tied pair, the outcome has a tied pair
    sig = [2 / 40.0, 2 / 40.0, 6 / 40.0, 9 / 40.0, 14 / 40.0]
    out = [0, 55, 0, 70, 100]
    blocks = blockset_by_key(sig, out, [0, 0, 0, 0, 0])
    b = blocks[0]
    allv = enum_T_values(blocks)
    # group arrangements by the resulting x2 vector and by T
    byx = {}
    byt = {}
    for perm, v in allv:
        xv = tuple(b.x2[k] for k in perm[0])
        byx.setdefault(xv, []).append(v)
        byt.setdefault(v, []).append(xv)
    # (a) arrangements that only swap the tied signal pair give byte-equal T
    same_x_exact = all(len(set(vs)) == 1 for vs in byx.values())
    # (b) two arrangements with DIFFERENT x2 vectors that give the same T
    pair = None
    for v, xvs in byt.items():
        dis = sorted(set(xvs))
        if len(dis) >= 2:
            pair = (v, dis[0], dis[1])
            break
    log("CHECK 2  tolerance rule and tied ranks, one block of 5 with ties")
    log("  signal midranks x2      : %s" % (b.x2,))
    log("  outcome midranks y2     : %s" % (b.y2,))
    log("  arrangements            : %d      distinct x2 vectors: %d" %
        (len(allv), len(byx)))
    log("  distinct T values       : %d" % len(byt))
    log("  tie-swap arrangements give byte-equal T : %s" %
        ("YES" if same_x_exact else "NO"))
    if pair is None:
        log("  different x2 with equal T : NONE FOUND")
        return False
    v, x1, x2v = pair
    d = abs(v - v)
    log("  different x2 with equal T : %s and %s" % (x1, x2v))
    log("    T both                  : %.17g      difference %.17g" % (v, d))
    log("    counted as equal by |T*| >= |T| - 1e-9 : %s" %
        ("YES" if d <= TOL else "NO"))
    # and the p-value counts every one of them
    lim = abs(v) - TOL
    cnt = sum(1 for _, vv in allv if abs(vv) >= lim)
    cnt_eq = sum(1 for _, vv in allv if abs(abs(vv) - abs(v)) <= TOL)
    log("  arrangements with |T*| >= |T|-tol at that T : %d   of which equal: %d" %
        (cnt, cnt_eq))
    return same_x_exact and d <= TOL and cnt_eq >= 2


def check_var_rho(log):
    """Var(rho_r) under S equals 1/(n_r - 1) by enumeration, with ties."""
    cases = [
        ("n=5, ties in both", [1 / 20.0, 1 / 20.0, 3 / 20.0, 7 / 20.0, 9 / 20.0],
         [0, 100, 0, 44, 100]),
        ("n=6, heavy ties", [0.0, 0.0, 2 / 25.0, 2 / 25.0, 5 / 25.0, 11 / 25.0],
         [0, 0, 100, 100, 37, 0]),
        ("n=4, no ties", [1 / 30.0, 4 / 30.0, 8 / 30.0, 15 / 30.0], [0, 20, 60, 100]),
    ]
    ok = True
    log("CHECK 3  Var(rho_r) under S equals 1/(n_r - 1), full enumeration")
    for label, sig, out in cases:
        blocks = blockset_by_key(sig, out, [0] * len(sig))
        b = blocks[0]
        allv = enum_T_values(blocks)
        rhos = [v / float(b.w) for _, v in allv]
        mean = sum(rhos) / len(rhos)
        var = sum((x - mean) ** 2 for x in rhos) / len(rhos)
        target = 1.0 / (b.n - 1)
        good = abs(var - target) < 1e-12 and abs(mean) < 1e-12
        ok = ok and good
        log("  %-20s n=%d  mean rho %.3e  Var %.15f  1/(n-1) %.15f  %s" %
            (label, b.n, mean, var, target, "YES" if good else "NO"))
    return ok

def mean_sd(xs):
    n = len(xs)
    if n == 0:
        return float("nan"), float("nan")
    m = sum(xs) / n
    if n < 2:
        return m, 0.0
    v = sum((x - m) ** 2 for x in xs) / (n - 1)
    return m, math.sqrt(v)


def mcerr(p, n):
    return math.sqrt(max(p * (1 - p), 1e-12) / n)


def quants(xs):
    s = sorted(xs)
    n = len(s)
    if n == 0:
        return (float("nan"),) * 5

    def q(f):
        i = int(f * (n - 1))
        return s[i]
    return (s[0], q(0.25), q(0.5), q(0.75), s[-1])

RHO_STAR = 0.15


def summarise_scen(scen, rows, truth_mu, tests, log):
    rb = [r["rho_bar"] for r in rows]
    m_rb, s_rb = mean_sd(rb)
    n = len(rows)
    cov = sum(1 for r in rows if r["lo"] <= truth_mu <= r["hi"]) / float(n)
    inside = sum(1 for r in rows if r["lo"] > -RHO_STAR and r["hi"] < RHO_STAR) / float(n)
    floorb = sum(r["floor_binds"] for r in rows) / float(n)
    mm, _ = mean_sd([r["m"] for r in rows])
    mw, _ = mean_sd([float(r["W"]) for r in rows])
    rates = {}
    for t in tests:
        rates[t] = sum(1 for r in rows if r[t] <= 0.05) / float(n)
    log("%-4s n=%4d  m %.1f  W %.1f  rho_bar mean %+.4f sd %.4f  true mean %+.4f" %
        (scen, n, mm, mw, m_rb, s_rb, truth_mu))
    log("      rejection at alpha 0.05:  " +
        "  ".join("%s %.3f(+-%.3f)" % (t, rates[t], mcerr(rates[t], n)) for t in tests))
    log("      interval: coverage of true mean %.3f   entirely inside (-0.15,+0.15) %.3f"
        "   floor binds %.3f" % (cov, inside, floorb))
    return {"scen": scen, "n": n, "rates": rates, "mean": m_rb, "sd": s_rb,
            "truth": truth_mu, "cov": cov, "inside": inside, "floor": floorb,
            "m": mm, "W": mw}
