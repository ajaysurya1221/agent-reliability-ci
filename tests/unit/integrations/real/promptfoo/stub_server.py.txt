"""Local HTTP stub for promptfoo's http provider (cache fixtures). Answers {"output": "the word is W"}.
Counts requests so the cache's effect is visible. Usage: python3 stub_server.py PORT COUNTFILE"""
import json, sys
from http.server import BaseHTTPRequestHandler, HTTPServer
port, countfile = int(sys.argv[1]), sys.argv[2]
class H(BaseHTTPRequestHandler):
    def do_POST(self):
        body = json.loads(self.rfile.read(int(self.headers.get("Content-Length", 0))) or b"{}")
        with open(countfile, "a") as f: f.write("1\n")
        out = json.dumps({"output": f"the word is {body.get('word', '')}"}).encode()
        self.send_response(200); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(out))); self.end_headers(); self.wfile.write(out)
    def log_message(self, *a): pass
HTTPServer(("127.0.0.1", port), H).serve_forever()
