"""Twenty tasks for the DeepEval fixtures. AGENT_MODE=pass|fail. Explicit, stable test ids."""
import os

import pytest
from deepeval import assert_test
from deepeval.metrics import BaseMetric
from deepeval.test_case import LLMTestCase

MODE = os.environ.get("AGENT_MODE", "pass")
WORDS = [f"w{n:02d}" for n in range(1, 21)]


def agent(word: str) -> str:
    return "nope" if MODE == "fail" else f"the word is {word}"


class ContainsWord(BaseMetric):
    """A code-only metric: no model is called."""

    def __init__(self) -> None:
        self.threshold = 1.0
        self.async_mode = False

    def measure(self, test_case, *args, **kwargs):
        self.score = 1.0 if test_case.expected_output in test_case.actual_output else 0.0
        self.reason = "contains the word" if self.score else "word missing"
        self.success = self.score >= self.threshold
        return self.score

    async def a_measure(self, test_case, *args, **kwargs):
        return self.measure(test_case)

    def is_successful(self) -> bool:
        return bool(self.success)

    @property
    def __name__(self):
        return "Contains Word"


@pytest.mark.parametrize("word", WORDS, ids=[f"task-{w}" for w in WORDS])
def test_agent(word):
    case = LLMTestCase(input=f"Say the word {word}", actual_output=agent(word), expected_output=word)
    assert_test(case, [ContainsWord()], run_async=False)
