"""MLflow 3.18.0 fixture generator. Usage: python evaluate_agent.py MODE OUTDIR  (MODE pass|fail).

Runs mlflow.genai.evaluate on 20 items with a local SQLite store, a code-only scorer, one item
whose predict_fn raises and one item whose scorer raises. Writes result_df (without the trace
objects) as JSON so the fixture shows the exact columns MLflow produces.
"""
import json
import sys
from pathlib import Path

import mlflow
from mlflow.genai.scorers import scorer

MODE, OUT = sys.argv[1], Path(sys.argv[2])
OUT.mkdir(parents=True, exist_ok=True)
mlflow.set_tracking_uri(f"sqlite:///{OUT / 'mlflow.db'}")
mlflow.set_experiment(f"arci-fixture-{MODE}")
WORDS = [f"w{n:02d}" for n in range(1, 21)]


def predict_fn(word: str) -> str:
    if word == "w07":
        raise RuntimeError("agent crashed")
    return "nope" if MODE == "fail" else f"the word is {word}"


@scorer
def contains_word(inputs, outputs, expectations) -> bool:
    if inputs["word"] == "w13":
        raise ValueError("scorer crashed")
    return expectations["word"] in str(outputs)


data = [{"inputs": {"word": w}, "expectations": {"word": w}} for w in WORDS]
res = mlflow.genai.evaluate(data=data, predict_fn=predict_fn, scorers=[contains_word])
df = res.result_df.drop(columns=[c for c in res.result_df.columns if c == "trace"])
(OUT / "result_df.json").write_text(df.to_json(orient="records", indent=1, default_handler=str))
(OUT / "summary.json").write_text(json.dumps({"run_id": res.run_id, "columns": list(res.result_df.columns), "passed": bool(getattr(res, "passed", False))}, indent=1))
print("run", res.run_id, "columns", list(res.result_df.columns))
