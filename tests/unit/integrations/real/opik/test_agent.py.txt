"""Twenty tasks for the Opik fixtures, with @opik.llm_unit. AGENT_MODE=pass|fail. Explicit ids."""
import os

import opik
import pytest

MODE = os.environ.get("AGENT_MODE", "pass")
WORDS = [f"w{n:02d}" for n in range(1, 21)]


@opik.track
def agent(word: str) -> str:
    return "nope" if MODE == "fail" else f"the word is {word}"


@opik.llm_unit()
@pytest.mark.parametrize("word", WORDS, ids=[f"task-{w}" for w in WORDS])
def test_agent(word):
    assert word in agent(word)
